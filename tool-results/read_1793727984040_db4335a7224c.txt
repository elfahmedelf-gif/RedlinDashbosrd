   267→function PermissionMatrix({ selected, onChange, disabled }: { selected: string[]; onChange: (perms: string[]) => void; disabled?: boolean }) {
   268→  const set = new Set(selected)
   269→  const toggle = (key: string) => {
   270→    if (disabled) return
   271→    const next = new Set(set)
   272→    if (next.has(key)) next.delete(key)
   273→    else next.add(key)
   274→    onChange(Array.from(next))
   275→  }
   276→  const toggleGroup = (group: string) => {
   277→    if (disabled) return
   278→    const keys = PERMISSION_CATALOG.filter((p) => p.group === group).map((p) => p.key)
   279→    const allOn = keys.every((k) => set.has(k))
   280→    const next = new Set(set)
   281→    keys.forEach((k) => (allOn ? next.delete(k) : next.add(k)))
   282→    onChange(Array.from(next))
   283→  }
   284→
   285→  return (
   286→    <div className="space-y-3 max-h-[50vh] overflow-y-auto pl-1 scrollbar-thin">
   287→      {PERMISSION_GROUPS.map((group) => {
   288→        const defs = PERMISSION_CATALOG.filter((p) => p.group === group)
   289→        const onCount = defs.filter((d) => set.has(d.key)).length
   290→        const allOn = onCount === defs.length
   291→        return (
   292→          <div key={group} className="rounded-xl border border-border overflow-hidden">
   293→            <div className="flex items-center justify-between gap-2 bg-muted/50 px-3 py-2">
   294→              <div
   295→                onClick={() => toggleGroup(group)}
   296→                role="button"
   297→                tabIndex={disabled ? -1 : 0}
   298→                onKeyDown={(e) => { if (!disabled && (e.key === 'Enter' || e.key === ' ')) { e.preventDefault(); toggleGroup(group) } }}
   299→                className={cn('flex items-center gap-2 text-xs font-bold hover:text-primary transition-colors cursor-pointer', disabled && 'opacity-60 cursor-not-allowed')}
   300→              >
   301→                <Checkbox checked={allOn} disabled={disabled} onCheckedChange={() => toggleGroup(group)} aria-label={`تحديد كل صلاحيات ${group}`} />
   302→                <span>{group}</span>
   303→              </div>
   304→              <Badge variant="secondary" className="text-[10px] tnum">{onCount}/{defs.length}</Badge>
   305→            </div>
   306→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-x-3 p-2.5">
   307→              {defs.map((d) => (
   308→                <label key={d.key} className={cn('flex items-center gap-2 py-1 px-1.5 rounded-lg text-xs cursor-pointer hover:bg-muted/60', disabled && 'opacity-60 cursor-not-allowed')}>
   309→                  <Checkbox checked={set.has(d.key)} disabled={disabled} onCheckedChange={() => toggle(d.key)} />
   310→                  <span className="flex-1 leading-tight">{d.label}</span>
   311→                </label>
   312→              ))}
   313→            </div>
   314→          </div>
   315→        )
   316→      })}
   317→    </div>
   318→  )
   319→}

   321→export function RolesModule() {

   362→        await apiPatch(`/api/roles/${editing.id}`, payload)
   363→        toast({ title: 'تم تحديث الدور', description: `${v.permissions.length} صلاحية محفوظة` })
   364→      } else {
   365→        await apiPost('/api/roles', payload)
   366→        toast({ title: 'تمت إضافة الدور', description: `${v.permissions.length} صلاحية` })
   367→      }
   368→      setOpen(false); mutate()
   369→    } catch (e: any) { setErr(e.message) } finally { setSaving(false) }
   370→  }
   371→
   372→  const delRole = async () => {
   373→    if (!deleting) return
   374→    try { await apiDelete(`/api/roles/${deleting.id}`); mutate(); toast({ title: 'تم حذف الدور' }) }
   375→    catch (e: any) { toast({ title: 'خطأ', description: e.message, variant: 'destructive' }) }
   376→    setDeleting(null)

   380→    <div className="space-y-4">
   381→      <div className="flex items-center justify-between gap-2 flex-wrap">
   382→        <div className="flex items-center gap-2 flex-wrap">
   383→          <Shield className="size-5 text-muted-foreground" />

   385→          <Badge variant="outline" className="tnum">{fmtNum(PERMISSION_CATALOG.length)} صلاحية في النظام</Badge>

   388→          <Plus className="size-4" /> دور جديد
   389→        </Button>
   390→      </div>
   391→
   392→      {isLoading ? (
   393→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   394→          {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-40 rounded-xl" />)}
   395→        </div>

   397→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   398→          {roles.map((r) => {

   494→export function SettingsModule() {
   495→  const { toast } = useToast()
   496→  const { user: me } = useAuth()
   497→  const { data, isLoading, mutate } = useSWR<{ items: any[] }>('/api/settings', fetcher)
   498→  const [drafts, setDrafts] = React.useState<Record<string, string>>({})
   499→  const [saving, setSaving] = React.useState(false)
   500→
   501→  React.useEffect(() => {
   502→    if (data && Object.keys(drafts).length === 0) {
   503→      const d: Record<string, string> = {}
   504→      for (const s of data.items) d[s.key] = s.value || ''
   505→      setDrafts(d)
   506→    }
   507→  }, [data])
   508→
   509→  const grouped = React.useMemo(() => {
   510→    if (!data) return [] as [string, any[]][]
   511→    const m = new Map<string, any[]>()
   512→    for (const s of data.items) {
   513→      const cat = s.category || 'general'
   514→      if (!m.has(cat)) m.set(cat, [])
   515→      m.get(cat)!.push(s)
   516→    }
   517→    return Array.from(m.entries())
   518→  }, [data])
   519→
   520→  const save = async () => {
   521→    setSaving(true)
   522→    try {
   523→      const updates = data!.items.map((s) => ({ key: s.key, value: drafts[s.key] ?? '', label: s.label, category: s.category }))
   524→      await apiPatch('/api/settings', { updates, userName: me?.username || 'admin' })
   525→      toast({ title: 'تم حفظ الإعدادات' })
   526→      mutate()
   527→    } catch (e: any) { toast({ title: 'خطأ', description: e.message, variant: 'destructive' }) }
   528→    finally { setSaving(false) }
   529→  }
   530→
   531→  const catLabels: Record<string, string> = {
   532→    general: 'عام', import: 'تعريفات أعمدة الاستيراد', accounting: 'محاسبي', inventory: 'المخزون',
   533→  }
   534→
   535→  return (
   536→    <div className="space-y-4">
   537→      <div className="flex items-center justify-between">
   538→        <div className="flex items-center gap-2"><SettingsIcon className="size-5 text-muted-foreground" /><span className="text-sm text-muted-foreground">إعدادات النظام</span></div>
   539→        <Button onClick={save} disabled={saving || isLoading} size="sm"><Save className="size-4" /> {saving ? 'جاري الحفظ...' : 'حفظ'}</Button>
   540→      </div>
   541→      {isLoading ? <Skeleton className="h-96" /> : (
   542→        <div className="space-y-4">
   543→          {grouped.map(([cat, items]) => (
   544→            <Card key={cat}>
   545→              <CardHeader className="pb-2"><CardTitle className="text-sm">{catLabels[cat] || cat}</CardTitle><CardDescription>{fmtNum(items.length)} إعداد</CardDescription></CardHeader>
   546→              <CardContent className="space-y-2.5">
   547→                {items.map((s) => (
   548→                  <div key={s.key} className="flex items-center gap-3">
   549→                    <div className="flex-1 min-w-0">
   550→                      <div className="text-sm font-medium">{s.label}</div>
   551→                      <div className="text-[10px] text-muted-foreground tnum">{s.key}</div>
   552→                    </div>
   553→                    {s.key === 'inventory_valuation' ? (
   554→                      <Select value={drafts[s.key] || ''} onValueChange={(val) => setDrafts((d) => ({ ...d, [s.key]: val }))}>
   555→                        <SelectTrigger className="w-56"><SelectValue /></SelectTrigger>
   556→                        <SelectContent>
   557→                          <SelectItem value="weighted_average">متوسط مرجح</SelectItem>
   558→                          <SelectItem value="fifo">الوارد أولًا يصرف أولًا (FIFO)</SelectItem>
   559→                          <SelectItem value="lifo">الوارد أخيرًا يصرف أولًا (LIFO)</SelectItem>
   560→                        </SelectContent>
   561→                      </Select>
   562→                    ) : (
   563→                      <Input value={drafts[s.key] || ''} onChange={(e) => setDrafts((d) => ({ ...d, [s.key]: e.target.value }))} className="w-64" dir={s.key.includes('label') || s.key === 'company_name' ? undefined : 'ltr'} />
   564→                    )}
   565→                  </div>
   566→                ))}
   567→              </CardContent>
   568→            </Card>
   569→          ))}
   570→        </div>
   571→      )}
   572→    </div>
   573→  )
   574→}
