    84→/** Demo-grade deterministic password hash (sync, no deps) — suitable for the in-memory mock system. */
    85→export function hashPassword(pwd: string): string {
    86→  let h1 = 0xdeadbeef, h2 = 0x41c6ce57
    87→  const s = `shipping::${pwd}::v1`
    88→  for (let i = 0; i < s.length; i++) {
    89→    const ch = s.charCodeAt(i)
    90→    h1 = Math.imul(h1 ^ ch, 2654435761)
    91→    h2 = Math.imul(h2 ^ ch, 1597334677)
    92→  }
    93→  h1 = Math.imul(h1 ^ (h1 >>> 16), 2246822507) ^ Math.imul(h2 ^ (h2 >>> 13), 3266489909)
    94→  h2 = Math.imul(h2 ^ (h2 >>> 16), 2246822507) ^ Math.imul(h1 ^ (h1 >>> 13), 3266489909)
    95→  return (4294967296 * (2097151 & h2) + (h1 >>> 0)).toString(36) + '.' + (h1 >>> 0).toString(36) + (h2 >>> 0).toString(36)
    96→}

    98→/** Convert epoch-millis numbers to ISO strings so date formatters work. */
    99→function isoDates<T extends Record<string, any>>(obj: T, keys: string[]): T {
   100→  for (const k of keys) {
   101→    const v = (obj as any)[k]
   102→    if (typeof v === 'number' && v > 1e11) (obj as any)[k] = new Date(v).toISOString()
   103→  }
   104→  return obj
   105→}

   415→function enrichShipment(s: any) {
   416→  const st = { ...s }
   417→  st.customer = partyRef(s.customerId)
   418→  st.contractor = partyRef(s.contractorId)
   419→  st.branch = s.branchId ? (branchById.get(s.branchId) ? { id: s.branchId, name: branchById.get(s.branchId).name } : null) : null
   420→  st.warehouse = s.warehouseId ? (whById.get(s.warehouseId) ? { id: s.warehouseId, name: whById.get(s.warehouseId).name } : null) : null
   421→  if (!Array.isArray(st.statusHistory)) {
   422→    st.statusHistory = [{
   423→      status: s.status,
   424→      notes: 'استيراد أولي من ملف مبيعات العاشر 3',
   425→      date: st.updatedAt ?? st.createdAt ?? null,
   426→      userName: 'system',
   427→    }]
   428→  }
   429→  // Profit margin = customer freight (sell) - contractor freight (cost)
   430→  const sell = Number(st.freightTotal ?? 0)
   431→  const cost = Number(st.totalSuppliers ?? 0)
   432→  st.margin = Math.round((sell - cost) * 100) / 100
   433→  // Linked vouchers (operational model: items-based linkage)
   434→  st.vouchers = db.vouchers
   435→    .filter((v: any) => Array.isArray(v.items) && v.items.some((it: any) => it.shipmentId === s.id))
   436→    .map((v: any) => ({ id: v.id, voucherNo: v.voucherNo, type: v.type, status: v.status, date: v.createdAt, amount: v.amount, notes: v.notes ?? null }))
   437→    .sort((a: any, b: any) => new Date(b.date ?? 0).getTime() - new Date(a.date ?? 0).getTime())
   438→  return st
   439→}

   538→function itemBalance(itemId: string): number {
   539→  let bal = 0
   540→  for (const m of db.movements) {
   541→    if (m.itemId !== itemId) continue
   542→    if (m.type === 'receipt') bal += m.quantity
   543→    else if (m.type === 'issue') bal += m.quantity // stored negative
   544→    else if (m.type === 'adjustment') bal += m.quantity // signed
   545→    // transfer keeps total balance
   546→  }
   547→  return Math.round(bal * 100) / 100
   548→}

   571→/** Account balances derived from posted journal lines. */
   572→function accountBalances(): Map<string, { debit: number; credit: number }> {
   573→  const map = new Map<string, { debit: number; credit: number }>()
   574→  for (const e of db.journal) {
   575→    if (e.status !== 'posted') continue
   576→    for (const l of e.lines) {
   577→      if (!l.accountId) continue
   578→      const cur = map.get(l.accountId) ?? { debit: 0, credit: 0 }
   579→      cur.debit += l.debit || 0
   580→      cur.credit += l.credit || 0
   581→      map.set(l.accountId, cur)
   582→    }
   583→  }
   584→  return map
   585→}

   587→function enrichAccount(a: any, balances: Map<string, { debit: number; credit: number }>) {
   588→  const b = balances.get(a.id) ?? { debit: 0, credit: 0 }
   589→  const parent = a.parentId ? db.accounts.find((x: any) => x.id === a.parentId) : null
   590→  return { ...a, parent: parent ? { id: parent.id, name: parent.name } : null, debit: b.debit, credit: b.credit, balance: b.debit - b.credit }
   591→}

   593→/** Deterministic party aggregates from shipments + vouchers. */
   594→function partyAggregates(): Map<string, { shipments: number; freightTotal: number; totalSuppliers: number; customerPay: number }> {
   595→  const map = new Map<string, any>()
   596→  const get = (id: string) => {
   597→    if (!map.has(id)) map.set(id, { shipments: 0, freightTotal: 0, totalSuppliers: 0, customerPay: 0 })
   598→    return map.get(id)
   599→  }
   600→  for (const s of db.shipments) {
   601→    if (s.customerId) {
   602→      const a = get(s.customerId)
   603→      a.shipments += 1
   604→      a.freightTotal += s.freightTotal ?? 0
   605→      a.totalSuppliers += s.totalSuppliers ?? 0
   606→      a.customerPay += s.customerPay ?? 0
   607→    }
   608→    if (s.contractorId && s.contractorId !== s.customerId) {
   609→      const a = get(s.contractorId)
   610→      a.shipments += 1
   611→      a.freightTotal += s.freightTotal ?? 0
   612→      a.totalSuppliers += s.totalSuppliers ?? 0
   613→    }
   614→  }
   615→  return map
   616→}

   625→function paginate<T>(arr: T[], page: number, pageSize: number) {
   626→  const total = arr.length
   627→  const totalPages = Math.max(1, Math.ceil(total / pageSize))
   628→  const p = Math.min(Math.max(1, page), totalPages)
   629→  const start = (p - 1) * pageSize
   630→  return { items: arr.slice(start, start + pageSize), total, page: p, pageSize, totalPages }
   631→}

   633→function inRange(dateISO: string | null | undefined, from: string | null, to: string | null): boolean {
   634→  if (!from && !to) return true
   635→  if (!dateISO) return false
   636→  const t = new Date(dateISO).getTime()
   637→  if (from && t < new Date(from + 'T00:00:00').getTime()) return false
   638→  if (to && t > new Date(to + 'T23:59:59').getTime()) return false
   639→  return true
   640→}

   936→      const items = [...map.values()].map((c: any) => ({ ...c, due: c.total - c.paid })).sort((a: any, b: any) => b.due - a.due)
   937→      return { items }
   938→    }
   939→
   940→    if (type === 'trial-balance') {
   941→      const bal = accountBalances()
   942→      return { items: db.accounts.filter((a: any) => a.isDetail).map((a: any) => enrichAccount(a, bal)) }
   943→    }
   944→
   945→    if (type === 'inventory-movements') {
   946→      const items = db.movements.map(enrichMovement)
   947→        .filter((m: any) => inRange(m.date, from, to))
   948→        .sort((a: any, b: any) => new Date(b.date).getTime() - new Date(a.date).getTime())
   949→      return { items }
   950→    }
   951→
   952→    if (type === 'low-stock') {
   953→      return { items: db.items.map(enrichItem).filter((i: any) => i.isLowStock) }
   954→    }
   955→
   956→    return { items: [] }
   957→  }
   958→
   959→  // unknown route → empty response (never hit the network)
   960→  return { items: [], total: 0 }
   961→}

  1551→  // unknown mutation → optimistic ok

  1574→  const uname = String(username ?? '').trim()

  1578→  if (!u.active) throw new Error('هذا الحساب موقوف — راجع مدير النظام')

  1581→  if (!perms.length && role?.name === 'admin') perms.push('*')
