   266→function PermissionMatrix({ selected, onChange, disabled }: { selected: string[]; onChange: (perms: string[]) => void; disabled?: boolean }) {
   267→  const set = new Set(selected)
   268→  const toggle = (key: string) => {
   269→    if (disabled) return
   270→    const next = new Set(set)
   271→    if (next.has(key)) next.delete(key)
   272→    else next.add(key)
   273→    onChange(Array.from(next))
   274→  }
   275→  const toggleGroup = (group: string) => {
   276→    if (disabled) return
   277→    const keys = PERMISSION_CATALOG.filter((p) => p.group === group).map((p) => p.key)
   278→    const allOn = keys.every((k) => set.has(k))
   279→    const next = new Set(set)
   280→    keys.forEach((k) => (allOn ? next.delete(k) : next.add(k)))
   281→    onChange(Array.from(next))
   282→  }
   283→
   284→  return (
   285→    <div className="space-y-3 max-h-[50vh] overflow-y-auto pl-1 scrollbar-thin">
   286→      {PERMISSION_GROUPS.map((group) => {
   287→        const defs = PERMISSION_CATALOG.filter((p) => p.group === group)
   288→        const onCount = defs.filter((d) => set.has(d.key)).length
   289→        const allOn = onCount === defs.length
   290→        return (
   291→          <div key={group} className="rounded-xl border border-border overflow-hidden">
   292→            <div className="flex items-center justify-between gap-2 bg-muted/50 px-3 py-2">
   293→              <div
   294→                onClick={() => toggleGroup(group)}
   295→                role="button"
   296→                tabIndex={disabled ? -1 : 0}
   297→                onKeyDown={(e) => { if (!disabled && (e.key === 'Enter' || e.key === ' ')) { e.preventDefault(); toggleGroup(group) } }}
   298→                className={cn('flex items-center gap-2 text-xs font-bold hover:text-primary transition-colors cursor-pointer', disabled && 'opacity-60 cursor-not-allowed')}
   299→              >
   300→                <Checkbox checked={allOn} disabled={disabled} onCheckedChange={() => toggleGroup(group)} aria-label={`تحديد كل صلاحيات ${group}`} />
   301→                <span>{group}</span>
   302→              </div>
   303→              <Badge variant="secondary" className="text-[10px] tnum">{onCount}/{defs.length}</Badge>
   304→            </div>
   305→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-x-3 p-2.5">
   306→              {defs.map((d) => (
   307→                <label key={d.key} className={cn('flex items-center gap-2 py-1 px-1.5 rounded-lg text-xs cursor-pointer hover:bg-muted/60', disabled && 'opacity-60 cursor-not-allowed')}>
   308→                  <Checkbox checked={set.has(d.key)} disabled={disabled} onCheckedChange={() => toggle(d.key)} />
   309→                  <span className="flex-1 leading-tight">{d.label}</span>
   310→                </label>
   311→              ))}
   312→            </div>
   313→          </div>
   314→        )
   315→      })}
   316→    </div>
   317→  )
   318→}

   320→export function RolesModule() {

   366→      }
   367→      setOpen(false); mutate()
   368→    } catch (e: any) { setErr(e.message) } finally { setSaving(false) }
   369→  }

   371→  const delRole = async () => {
   372→    if (!deleting) return
   373→    try { await apiDelete(`/api/roles/${deleting.id}`); mutate(); toast({ title: 'تم حذف الدور' }) }
   374→    catch (e: any) { toast({ title: 'خطأ', description: e.message, variant: 'destructive' }) }

   390→
   391→      {isLoading ? (
   392→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   393→          {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-40 rounded-xl" />)}
   394→        </div>

   396→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">

   493→export function SettingsModule() {
   494→  const { toast } = useToast()
   495→  const { user: me } = useAuth()
   496→  const { data, isLoading, mutate } = useSWR<{ items: any[] }>('/api/settings', fetcher)
   497→  const [drafts, setDrafts] = React.useState<Record<string, string>>({})
   498→  const [saving, setSaving] = React.useState(false)
   499→
   500→  React.useEffect(() => {
   501→    if (data && Object.keys(drafts).length === 0) {
   502→      const d: Record<string, string> = {}
   503→      for (const s of data.items) d[s.key] = s.value || ''
   504→      setDrafts(d)
   505→    }
   506→  }, [data])
   507→
   508→  const grouped = React.useMemo(() => {
   509→    if (!data) return [] as [string, any[]][]
   510→    const m = new Map<string, any[]>()
   511→    for (const s of data.items) {
   512→      const cat = s.category || 'general'
   513→      if (!m.has(cat)) m.set(cat, [])
   514→      m.get(cat)!.push(s)
   515→    }
   516→    return Array.from(m.entries())
   517→  }, [data])
   518→
   519→  const save = async () => {
   520→    setSaving(true)
   521→    try {
   522→      const updates = data!.items.map((s) => ({ key: s.key, value: drafts[s.key] ?? '', label: s.label, category: s.category }))
   523→      await apiPatch('/api/settings', { updates, userName: me?.username || 'admin' })
   524→      toast({ title: 'تم حفظ الإعدادات' })
   525→      mutate()
   526→    } catch (e: any) { toast({ title: 'خطأ', description: e.message, variant: 'destructive' }) }
   527→    finally { setSaving(false) }
   528→  }
   529→
   530→  const catLabels: Record<string, string> = {
   531→    general: 'عام', import: 'تعريفات أعمدة الاستيراد', accounting: 'محاسبي', inventory: 'المخزون',
   532→  }
   533→
   534→  return (
   535→    <div className="space-y-4">
   536→      <div className="flex items-center justify-between">
   537→        <div className="flex items-center gap-2"><SettingsIcon className="size-5 text-muted-foreground" /><span className="text-sm text-muted-foreground">إعدادات النظام</span></div>
   538→        <Button onClick={save} disabled={saving || isLoading} size="sm"><Save className="size-4" /> {saving ? 'جاري الحفظ...' : 'حفظ'}</Button>
   539→      </div>
   540→      {isLoading ? <Skeleton className="h-96" /> : (
   541→        <div className="space-y-4">
   542→          {grouped.map(([cat, items]) => (
   543→            <Card key={cat}>
   544→              <CardHeader className="pb-2"><CardTitle className="text-sm">{catLabels[cat] || cat}</CardTitle><CardDescription>{fmtNum(items.length)} إعداد</CardDescription></CardHeader>
   545→              <CardContent className="space-y-2.5">
   546→                {items.map((s) => (
   547→                  <div key={s.key} className="flex items-center gap-3">
   548→                    <div className="flex-1 min-w-0">
   549→                      <div className="text-sm font-medium">{s.label}</div>
   550→                      <div className="text-[10px] text-muted-foreground tnum">{s.key}</div>
   551→                    </div>
   552→                    {s.key === 'inventory_valuation' ? (
   553→                      <Select value={drafts[s.key] || ''} onValueChange={(val) => setDrafts((d) => ({ ...d, [s.key]: val }))}>
   554→                        <SelectTrigger className="w-56"><SelectValue /></SelectTrigger>
   555→                        <SelectContent>
   556→                          <SelectItem value="weighted_average">متوسط مرجح</SelectItem>
   557→                          <SelectItem value="fifo">الوارد أولًا يصرف أولًا (FIFO)</SelectItem>
   558→                          <SelectItem value="lifo">الوارد أخيرًا يصرف أولًا (LIFO)</SelectItem>
   559→                        </SelectContent>
   560→                      </Select>
   561→                    ) : (
   562→                      <Input value={drafts[s.key] || ''} onChange={(e) => setDrafts((d) => ({ ...d, [s.key]: e.target.value }))} className="w-64" dir={s.key.includes('label') || s.key === 'company_name' ? undefined : 'ltr'} />
   563→                    )}
   564→                  </div>
   565→                ))}
   566→              </CardContent>
   567→            </Card>
   568→          ))}
   569→        </div>
   570→      )}
   571→    </div>
   572→  )
   573→}
