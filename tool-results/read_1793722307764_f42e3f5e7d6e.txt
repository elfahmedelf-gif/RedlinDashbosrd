   616→function ShipmentDetail({ id, onBack }: { id: string; onBack: () => void }) {
   617→  const { toast } = useToast()
   618→  const { data: shipment, mutate } = useSWR<any>(`/api/shipments/${id}`, fetcher)
   619→  const { items: statusItems } = useLookup('shipment_status')
   620→  const { items: collectionItems } = useLookup('collection_status')
   621→  const [statusOpen, setStatusOpen] = React.useState(false)
   622→  const [newStatus, setNewStatus] = React.useState('')
   623→  const [statusNotes, setStatusNotes] = React.useState('')
   624→  const [waybillOpen, setWaybillOpen] = React.useState(false)
   625→  const [codOpen, setCodOpen] = React.useState(false)
   626→  const [codCustomerAmount, setCodCustomerAmount] = React.useState('')
   627→  const [codContractorAmount, setCodContractorAmount] = React.useState('')
   628→
   629→  React.useEffect(() => {
   630→    if (shipment && codOpen) {
   631→      setCodCustomerAmount(String(shipment.freightTotal ?? 0))
   632→      setCodContractorAmount(String(shipment.totalSuppliers ?? shipment.freightCost ?? 0))
   633→    }
   634→  }, [shipment, codOpen])
   635→
   636→  if (!shipment) return <Skeleton className="h-96 rounded-xl" />
   637→
   638→  const STATUS_LOOKUP: Record<string, { label: string; tone: string }> = Object.fromEntries(
   639→    statusItems.map((i) => [i.key, { label: i.label, tone: i.tone || '' }]),
   640→  )
   641→  const COLLECTION_LOOKUP: Record<string, { label: string; tone: string }> = Object.fromEntries(
   642→    collectionItems.map((i) => [i.key, { label: i.label, tone: i.tone || '' }]),
   643→  )
   644→  const sMeta = STATUS_LOOKUP[shipment.status] || { label: shipment.status, tone: '' }
   645→  const cMeta = COLLECTION_LOOKUP[shipment.collectionStatus] || { label: shipment.collectionStatus, tone: '' }
   646→  const margin = (Number(shipment.freightTotal ?? 0)) - (Number(shipment.totalSuppliers ?? 0))
   647→
   648→  const changeStatus = async () => {
   649→    if (!newStatus) return

   651→      await apiPost(`/api/shipments/${id}/status`, { status: newStatus, notes: statusNotes, userName: 'admin' })
   652→      toast({ title: 'تم تحديث الحالة' })

   674→      await apiDelete(`/api/shipments/${id}`)
   675→      toast({ title: 'تم الحذف' })
   676→      onBack()
   677→    } catch (e: any) { toast({ title: 'خطأ', description: e.message, variant: 'destructive' }) }
   678→  }
   679→

   681→    <div className="space-y-4">
   682→      <div className="flex items-center justify-between">

   685→          <Button variant="outline" size="sm" onClick={() => setWaybillOpen(true)}><Printer className="size-4" /> طباعة بوليصة</Button>
   686→          <Button variant="outline" size="sm" onClick={() => setCodOpen(true)}><Banknote className="size-4" /> تسوية COD</Button>

   692→      <Card>
   693→        <CardContent className="p-5">
   694→          <div className="flex flex-col md:flex-row md:items-start md:justify-between gap-3">
   695→            <div>
   696→              <div className="text-xs text-muted-foreground">رقم الشحنة</div>
   697→              <div className="text-2xl font-bold font-mono">{shipment.shipmentNo}</div>
   698→              {shipment.billNo && <div className="text-sm text-muted-foreground tnum mt-1">بوليصة: {shipment.billNo}</div>}
   699→            </div>
   700→            <div className="flex gap-2 flex-wrap">
   701→              <Badge variant="outline" className={cn('font-bold', sMeta.tone)}>{sMeta.label}</Badge>
   702→              <Badge variant="outline" className={cn('font-bold', cMeta.tone)}>{cMeta.label}</Badge>
   703→            </div>
   704→          </div>

   711→          <div>

   735→            <Row label="الوجهة" value={shipment.destination || '-'} />
   736→            <Row label="الوجهة (EN)" value={shipment.destinationEn || '-'} ltr />
   737→            <Row label="خط السير/التكويد" value={shipment.routeCode || '-'} />
   738→          </CardContent>
   739→        </Card>
   740→        <Card>
   741→          <CardContent className="p-4 space-y-2 text-sm">
   742→            <h3 className="font-semibold mb-2">بيانات التشغيل</h3>

   744→            <Row label="المخزن" value={shipment.warehouse?.name || '-'} />

   746→            <Row label="نوع الحمولة" value={shipment.loadType || '-'} />
   747→            <Row label="رقم السيارة" value={shipment.vehicleNo || '-'} ltr />
   748→            <Row label="الكمية (سياسات)" value={fmtNum(shipment.policyQty)} />
   749→          </CardContent>
   750→        </Card>
   751→      </div>
   752→
   753→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   754→        <Card>
   755→          <CardContent className="p-4 space-y-2 text-sm">
   756→            <h3 className="font-semibold mb-2">المبالغ المالية</h3>
   757→            <Row label="تكلفة النولون" value={`${fmtNum(shipment.freightCost)} ج.م`} money />
   758→            <Row label="إجمالي النولون" value={`${fmtNum(shipment.freightTotal)} ج.م`} money bold />
   759→            <Row label="سعر البيع" value={`${fmtNum(shipment.salePrice ?? 0)} ج.م`} money />
   760→            <Row label="تحصيل العميل" value={`${fmtNum(shipment.customerPay)} ج.م`} money />
   761→            <Row label="إجمالي الموردين" value={`${fmtNum(shipment.totalSuppliers)} ج.م`} money bold />
   762→            <Row label="التعتيق" value={`${fmtNum(shipment.aging)} ج.م`} money />
   763→            <Row label="المبيت" value={`${fmtNum(shipment.overnight)} ج.م`} money />
   764→            <Row label="المرتجع" value={`${fmtNum(shipment.returns)} ج.م`} money />
   765→            <Row label="التحويلة" value={`${fmtNum(shipment.transfer)} ج.م`} money />
   766→            <Row label="أخرى / Drop" value={`${fmtNum((shipment.other ?? 0) + (shipment.drop ?? 0))} ج.م`} money />
   767→          </CardContent>
   768→        </Card>
   769→        <Card>
   770→          <CardContent className="p-4 space-y-2 text-sm">
   771→            <h3 className="font-semibold mb-2">التواريخ والملاحظات</h3>
   772→            <Row label="تاريخ الشحن" value={fmtDate(shipment.date)} />
   773→            <Row label="تاريخ (2)" value={fmtDate(shipment.date2)} />
   774→            <Row label="تاريخ التسليم الفعلي" value={fmtDate(shipment.actualDeliveryDate)} />
   775→            <Row label="ملاحظات" value={shipment.notes || '-'} />
   776→            <Row label="ملاحظة ص" value={shipment.note1 || '-'} />
   777→          </CardContent>
   778→        </Card>
   779→      </div>
   780→
   781→      {/* Linked vouchers */}
   782→      <Card>
   783→        <CardContent className="p-4">
   784→          <h3 className="font-semibold mb-3 text-sm flex items-center gap-2">
   785→            <Link2 className="size-4 text-muted-foreground" /> السندات المرتبطة بالشحنة
   786→          </h3>
   787→          {(shipment.vouchers ?? []).length === 0 ? (
   788→            <p className="text-sm text-muted-foreground">لا توجد سندات مرتبطة بهذه الشحنة. استخدم «تسوية COD» لتوليد سندات قبض/صرف تلقائية.</p>
   789→          ) : (
   790→            <div className="space-y-2">
   791→              {(shipment.vouchers ?? []).map((v: any) => (
   792→                <div key={v.id} className="flex items-center justify-between gap-3 text-sm border-b last:border-0 pb-2 last:pb-0">
   793→                  <div className="flex items-center gap-2">
   794→                    <Badge variant="outline" className={v.type === 'receipt' ? 'bg-emerald-50 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300' : 'bg-rose-50 text-rose-700 dark:bg-rose-900/40 dark:text-rose-300'}>
   795→                      {v.type === 'receipt' ? 'قبض' : 'صرف'}
   796→                    </Badge>
   797→                    <span className="font-mono text-xs font-bold">{v.voucherNo}</span>
   798→                    <span className="text-xs text-muted-foreground truncate-1">{v.description || '-'}</span>
   799→                  </div>
   800→                  <div className="flex items-center gap-2 shrink-0">
   801→                    <span className="tnum font-bold text-sm">{fmtNum(v.amount)} ج.م</span>
   802→                    <span className="text-[10px] text-muted-foreground tnum">{fmtDate(v.date)}</span>
   803→                  </div>
   804→                </div>
   805→              ))}
   806→            </div>
   807→          )}
   808→        </CardContent>
   809→      </Card>
   810→
   811→      {/* Status history */}
   812→      <Card>
   813→        <CardContent className="p-4">
   814→          <h3 className="font-semibold mb-3 text-sm flex items-center gap-2">
   815→            <Layers className="size-4 text-muted-foreground" /> سجل تغييرات الحالة
   816→          </h3>
   817→          {shipment.statusHistory?.length === 0 ? (
   818→            <p className="text-sm text-muted-foreground">لا يوجد سجل تغييرات.</p>
   819→          ) : (
   820→            <div className="space-y-2">
   821→              {(shipment.statusHistory ?? []).map((h: any, i: number) => {
   822→                const statusKey = h.toStatus ?? h.status
   823→                const statusLabel = STATUS_LOOKUP[statusKey]?.label || statusKey
   824→                const statusTone = STATUS_LOOKUP[statusKey]?.tone || ''
   825→                return (
   826→                  <div key={h.id ?? i} className="flex items-center gap-3 text-sm border-b last:border-0 pb-2 last:pb-0 flex-wrap">

   828→                    {h.fromStatus && <span className="text-muted-foreground text-xs">من {STATUS_LOOKUP[h.fromStatus]?.label || h.fromStatus}</span>}
   829→                    <span className="text-xs text-muted-foreground tnum">{fmtDateTime(h.changedAt ?? h.date)}</span>
   830→                    <span className="text-xs text-muted-foreground">— {h.userName}</span>
   831→                    {h.notes && <span className="text-xs text-muted-foreground">• {h.notes}</span>}
   832→                  </div>
   833→                )
   834→              })}
   835→            </div>
   836→          )}
   837→        </CardContent>
   838→      </Card>
   839→
   840→      {/* Status change dialog */}
   841→      <Dialog open={statusOpen} onOpenChange={setStatusOpen}>
   842→        <DialogContent>
   843→          <DialogHeader><DialogTitle>تغيير حالة الشحنة</DialogTitle></DialogHeader>
   844→          <div className="space-y-3 py-2">
   845→            <Select value={newStatus} onValueChange={setNewStatus}>
   846→              <SelectTrigger><SelectValue placeholder="اختر الحالة الجديدة..." /></SelectTrigger>
   847→              <SelectContent>
   848→                {statusItems.map((i) => <SelectItem key={i.key} value={i.key}>{i.label}</SelectItem>)}
   849→              </SelectContent>
   850→            </Select>
   851→            <Textarea value={statusNotes} onChange={(e) => setStatusNotes(e.target.value)} placeholder="ملاحظات (اختياري)..." rows={2} />
   852→          </div>
   853→          <DialogFooter>
   854→            <Button variant="outline" onClick={() => setStatusOpen(false)}>إلغاء</Button>
   855→            <Button onClick={changeStatus}>تأكيد</Button>
   856→          </DialogFooter>
   857→        </DialogContent>
   858→      </Dialog>
   859→
   860→      {/* COD settlement dialog */}
   861→      <Dialog open={codOpen} onOpenChange={setCodOpen}>
   862→        <DialogContent>
   863→          <DialogHeader>
   864→            <DialogTitle>تسوية الدفع عند الاستلام (COD)</DialogTitle>
   865→            <DialogDescription>
   866→              يصدر النظام سند قبض من العميل بقيمة نولون العميل وسند صرف للمقاول بقيمة نولون المقاول — مع قيود يومية تلقائية — ويُحدّث حالة التحصيل.
   867→            </DialogDescription>
   868→          </DialogHeader>
   869→          <div className="grid grid-cols-2 gap-3 py-2">
   870→            <div>
   871→              <label className="text-xs font-medium">سند قبض من العميل (ج.م)</label>
   872→              <Input type="number" value={codCustomerAmount} onChange={(e) => setCodCustomerAmount(e.target.value)} />
   873→              <p className="text-[10px] text-muted-foreground mt-0.5">قيمة نولون العميل — يصدر سند قبض للعميل ويُرحّل في الصندوق ومديونية العميل.</p>
   874→            </div>
   875→            <div>
   876→              <label className="text-xs font-medium">سند صرف للمقاول (ج.م)</label>
   877→              <Input type="number" value={codContractorAmount} onChange={(e) => setCodContractorAmount(e.target.value)} />
   878→              <p className="text-[10px] text-muted-foreground mt-0.5">قيمة نولون المقاول — يصدر سند صرف للمقاول ويُرحّل في الصندوق ومستحقات المقاولين.</p>
   879→            </div>
   880→          </div>
   881→          <div className="text-xs text-muted-foreground bg-muted/30 rounded-md p-2">
   882→            العميل: <b>{shipment.customer?.name || shipment.recipientName || '-'}</b> — المقاول: <b>{shipment.contractor?.name || '-'}</b>
   883→          </div>
   884→          <DialogFooter>
   885→            <Button variant="outline" onClick={() => setCodOpen(false)}>إلغاء</Button>
   886→            <Button onClick={settleCOD}><Banknote className="size-4" /> تأكيد التسوية</Button>
   887→          </DialogFooter>
   888→        </DialogContent>
   889→      </Dialog>
   890→
   891→      {/* Waybill print */}
   892→      <WaybillPrint shipment={shipment} open={waybillOpen} onOpenChange={setWaybillOpen} />
   893→    </div>
   894→  )
   895→}

   897→function Row({ label, value, ltr, money, bold }: { label: string; value: string; ltr?: boolean; money?: boolean; bold?: boolean }) {

   899→    <div className="flex items-center justify-between gap-3 py-1 border-b last:border-0">

   902→    </div>
   903→  )
   904→}
