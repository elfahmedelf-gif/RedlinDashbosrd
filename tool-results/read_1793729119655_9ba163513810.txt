    85→    85→export function hashPassword(pwd: string): string {
    86→    86→  let h1 = 0xdeadbeef, h2 = 0x41c6ce57
    87→    87→  const s = `shipping::${pwd}::v1`
    88→    88→  for (let i = 0; i < s.length; i++) {
    89→    89→    const ch = s.charCodeAt(i)
    90→    90→    h1 = Math.imul(h1 ^ ch, 2654435761)
    91→    91→    h2 = Math.imul(h2 ^ ch, 1597334677)
    92→    92→  }
    93→    93→  h1 = Math.imul(h1 ^ (h1 >>> 16), 2246822507) ^ Math.imul(h2 ^ (h2 >>> 13), 3266489909)
    94→    94→  h2 = Math.imul(h2 ^ (h2 >>> 16), 2246822507) ^ Math.imul(h1 ^ (h1 >>> 13), 3266489909)
    95→    95→  return (4294967296 * (2097151 & h2) + (h1 >>> 0)).toString(36) + '.' + (h1 >>> 0).toString(36) + (h2 >>> 0).toString(36)
    96→    96→}

    99→    99→function isoDates<T extends Record<string, any>>(obj: T, keys: string[]): T {
   100→   100→  for (const k of keys) {
   101→   101→    const v = (obj as any)[k]
   102→   102→    if (typeof v === 'number' && v > 1e11) (obj as any)[k] = new Date(v).toISOString()
   103→   103→  }
   104→   104→  return obj
   105→   105→}

   415→   415→function enrichShipment(s: any) {
   416→   416→  const st = { ...s }
   417→   417→  st.customer = partyRef(s.customerId)
   418→   418→  st.contractor = partyRef(s.contractorId)
   419→   419→  st.branch = s.branchId ? (branchById.get(s.branchId) ? { id: s.branchId, name: branchById.get(s.branchId).name } : null) : null
   420→   420→  st.warehouse = s.warehouseId ? (whById.get(s.warehouseId) ? { id: s.warehouseId, name: whById.get(s.warehouseId).name } : null) : null
   421→   421→  if (!Array.isArray(st.statusHistory)) {
   422→   422→    st.statusHistory = [{
   423→   423→      status: s.status,
   424→   424→      notes: 'استيراد أولي من ملف مبيعات العاشر 3',
   425→   425→      date: st.updatedAt ?? st.createdAt ?? null,
   426→   426→      userName: 'system',
   427→   427→    }]
   428→   428→  }
   429→   429→  // Profit margin = customer freight (sell) - contractor freight (cost)
   430→   430→  const sell = Number(st.freightTotal ?? 0)
   431→   431→  const cost = Number(st.totalSuppliers ?? 0)
   432→   432→  st.margin = Math.round((sell - cost) * 100) / 100
   433→   433→  // Linked vouchers (operational model: items-based linkage)
   434→   434→  st.vouchers = db.vouchers
   435→   435→    .filter((v: any) => Array.isArray(v.items) && v.items.some((it: any) => it.shipmentId === s.id))
