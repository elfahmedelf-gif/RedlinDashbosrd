   621→function ShipmentDetail({ id, onBack }: { id: string; onBack: () => void }) {

   625→  const { items: collectionItems } = useLookup('collection_status')

   628→  const [statusNotes, setStatusNotes] = React.useState('')

   632→  const [codContractorAmount, setCodContractorAmount] = React.useState('')

   741→            <Row label="الوجهة (EN)" value={shipment.destinationEn || '-'} ltr />
   742→            <Row label="خط السير/التكويد" value={shipment.routeCode || '-'} />
   743→          </CardContent>
   744→        </Card>
   745→        <Card>
   746→          <CardContent className="p-4 space-y-2 text-sm">
   747→            <h3 className="font-semibold mb-2">بيانات التشغيل</h3>
   748→            <Row label="الفرع" value={shipment.branch?.name || '-'} />
   749→            <Row label="المخزن" value={shipment.warehouse?.name || '-'} />
   750→            <Row label="المقاول" value={shipment.contractor?.name || '-'} />
   751→            <Row label="نوع الحمولة" value={shipment.loadType || '-'} />
   752→            <Row label="رقم السيارة" value={shipment.vehicleNo || '-'} ltr />
   753→            <Row label="الكمية (سياسات)" value={fmtNum(shipment.policyQty)} />
   754→          </CardContent>
   755→        </Card>
   756→      </div>
   757→
   758→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   759→        <Card>
   760→          <CardContent className="p-4 space-y-2 text-sm">
   761→            <h3 className="font-semibold mb-2">المبالغ المالية</h3>
   762→            <Row label="تكلفة النولون" value={`${fmtNum(shipment.freightCost)} ج.م`} money />
   763→            <Row label="إجمالي النولون" value={`${fmtNum(shipment.freightTotal)} ج.م`} money bold />
   764→            <Row label="سعر البيع" value={`${fmtNum(shipment.salePrice ?? 0)} ج.م`} money />
   765→            <Row label="تحصيل العميل" value={`${fmtNum(shipment.customerPay)} ج.م`} money />
   766→            <Row label="إجمالي الموردين" value={`${fmtNum(shipment.totalSuppliers)} ج.م`} money bold />
   767→            <Row label="التعتيق" value={`${fmtNum(shipment.aging)} ج.م`} money />
   768→            <Row label="المبيت" value={`${fmtNum(shipment.overnight)} ج.م`} money />
   769→            <Row label="المرتجع" value={`${fmtNum(shipment.returns)} ج.م`} money />
   770→            <Row label="التحويلة" value={`${fmtNum(shipment.transfer)} ج.م`} money />
   771→            <Row label="أخرى / Drop" value={`${fmtNum((shipment.other ?? 0) + (shipment.drop ?? 0))} ج.م`} money />
   772→          </CardContent>
   773→        </Card>
   774→        <Card>
   775→          <CardContent className="p-4 space-y-2 text-sm">
   776→            <h3 className="font-semibold mb-2">التواريخ والملاحظات</h3>
   777→            <Row label="تاريخ الشحن" value={fmtDate(shipment.date)} />
   778→            <Row label="تاريخ (2)" value={fmtDate(shipment.date2)} />
   779→            <Row label="تاريخ التسليم الفعلي" value={fmtDate(shipment.actualDeliveryDate)} />
   780→            <Row label="ملاحظات" value={shipment.notes || '-'} />
   781→            <Row label="ملاحظة ص" value={shipment.note1 || '-'} />
   782→          </CardContent>
   783→        </Card>
   784→      </div>
   785→
   786→      {/* Linked vouchers */}
   787→      <Card>
   788→        <CardContent className="p-4">
   789→          <h3 className="font-semibold mb-3 text-sm flex items-center gap-2">
   790→            <Link2 className="size-4 text-muted-foreground" /> السندات المرتبطة بالشحنة
   791→          </h3>
   792→          {(shipment.vouchers ?? []).length === 0 ? (
   793→            <p className="text-sm text-muted-foreground">لا توجد سندات مرتبطة بهذه الشحنة. استخدم «تسوية COD» لتوليد سندات قبض/صرف تلقائية.</p>
   794→          ) : (
   795→            <div className="space-y-2">
   796→              {(shipment.vouchers ?? []).map((v: any) => (
   797→                <div key={v.id} className="flex items-center justify-between gap-3 text-sm border-b last:border-0 pb-2 last:pb-0">
   798→                  <div className="flex items-center gap-2">
   799→                    <Badge variant="outline" className={v.type === 'receipt' ? 'bg-emerald-50 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300' : 'bg-rose-50 text-rose-700 dark:bg-rose-900/40 dark:text-rose-300'}>
   800→                      {v.type === 'receipt' ? 'قبض' : 'صرف'}
   801→                    </Badge>
   802→                    <span className="font-mono text-xs font-bold">{v.voucherNo}</span>
   803→                    <span className="text-xs text-muted-foreground truncate-1">{v.description || '-'}</span>
   804→                  </div>
   805→                  <div className="flex items-center gap-2 shrink-0">
   806→                    <span className="tnum font-bold text-sm">{fmtNum(v.amount)} ج.م</span>
   807→                    <span className="text-[10px] text-muted-foreground tnum">{fmtDate(v.date)}</span>
   808→                  </div>
   809→                </div>
   810→              ))}
   811→            </div>
   812→          )}
   813→        </CardContent>
   814→      </Card>
   815→
   816→      {/* Status history */}
   817→      <Card>
   818→        <CardContent className="p-4">
   819→          <h3 className="font-semibold mb-3 text-sm flex items-center gap-2">
   820→            <Layers className="size-4 text-muted-foreground" /> سجل تغييرات الحالة
   821→          </h3>
   822→          {shipment.statusHistory?.length === 0 ? (
   823→            <p className="text-sm text-muted-foreground">لا يوجد سجل تغييرات.</p>
   824→          ) : (
   825→            <div className="space-y-2">
   826→              {(shipment.statusHistory ?? []).map((h: any, i: number) => {
   827→                const statusKey = h.toStatus ?? h.status
   828→                const statusLabel = STATUS_LOOKUP[statusKey]?.label || statusKey
   829→                const statusTone = STATUS_LOOKUP[statusKey]?.tone || ''
   830→                return (
   831→                  <div key={h.id ?? i} className="flex items-center gap-3 text-sm border-b last:border-0 pb-2 last:pb-0 flex-wrap">
   832→                    <Badge variant="outline" className={statusTone}>{statusLabel}</Badge>
   833→                    {h.fromStatus && <span className="text-muted-foreground text-xs">من {STATUS_LOOKUP.fromStatus]?.label || h.fromStatus}</span>}
   834→                    <span className="text-xs text-muted-foreground tnum">{fmtDateTime(h.changedAt ?? h.date)}</span>
   835→                    <span className="text-xs text-muted-foreground">— {h.userName}</span>
   836→                    {h.notes && <span className="text-xs text-muted-foreground">• {h.notes}</span>}
   837→                  </div>
   838→                )
   839→              })}
   840→            </div>
   841→          )}
   842→        </CardContent>
   843→      </Card>
   844→
   845→      {/* Status change dialog */}
   846→      <Dialog open={statusOpen} onOpenChange={setStatusOpen}>
   847→        <DialogContent>
   848→          <DialogHeader><DialogTitle>تغيير حالة الشحنة</DialogTitle></DialogHeader>
   849→          <div className="space-y-3 py-2">
   850→            <Select value={newStatus} onValueChange={setNewStatus}>
   851→              <SelectTrigger><SelectValue placeholder="اختر الحالة الجديدة..." /></SelectTrigger>
   852→              <SelectContent>
   853→                {statusItems.map((i) => <SelectItem key={i.key} value={i.key}>{i.label}</SelectItem>)}
   854→              </SelectContent>
   855→            </Select>
   856→            <Textarea value={statusNotes} onChange={(e) => setStatusNotes(e.target.value)} placeholder="ملاحظات (اختياري)..." rows={2} />
   857→          </div>
   858→          <DialogFooter>
   859→            <Button variant="outline" onClick={() => setStatusOpen(false)}>إلغاء</Button>
   860→            <Button onClick={changeStatus}>تأكيد</Button>
   861→          </DialogFooter>
   862→        </DialogContent>
   863→      </Dialog>
   864→
   865→      {/* COD settlement dialog */}
   866→      <Dialog open={codOpen} onOpenChange={setCodOpen}>
   867→        <DialogContent>
   868→          <DialogHeader>
   869→            <DialogTitle>تسوية الدفع عند الاستلام (COD)</DialogTitle>
   870→            <DialogDescription>
   871→              يصدر النظام سند قبض من العميل بقيمة نولون العميل وسند صرف للمقاول بقيمة نولون المقاول — مع قيود يومية تلقائية — ويُحدّث حالة التحصيل.
   872→            </DialogDescription>
   873→          </DialogHeader>
   874→          <div className="grid grid-cols-2 gap-3 py-2">
   875→            <div>
   876→              <label className="text-xs font-medium">سند قبض من العميل (ج.م)</label>
   877→              <Input type="number" value={codCustomerAmount} onChange={(e) => setCodCustomerAmount(e.target.value)} />
   878→              <p className="text-[10px] text-muted-foreground mt-0.5">قيمة نولون العميل — يصدر سند قبض للعميل ويُرحّل في الصندوق ومديونية العميل.</p>
   879→            </div>
   880→            <div>
   881→              <label className="text-xs font-medium">سند صرف للمقاول (ج.م)</label>
   882→              <Input type="number" value={codContractorAmount} onChange={(e) => setCodContractorAmount(e.target.value)} />
   883→              <p className="text-[10px] text-muted-foreground mt-0.5">قيمة نولون المقاول — يصدر سند صرف للمقاول ويُرحّل في الصندوق ومستحقات المقاولين.</p>
   884→            </div>
   885→          </div>
   886→          <div className="text-xs text-muted-foreground bg-muted/30 rounded-md p-2">
   887→            العميل: <b>{shipment.customer?.name || shipment.recipientName || '-'}</b> — المقاول: <b>{shipment.contractor?.name || '-'}</b>
   888→          </div>
   889→          <DialogFooter>
   890→            <Button variant="outline" onClick={() => setCodOpen(false)}>إلغاء</Button>
   891→            <Button onClick={settleCOD}><Banknote className="size-4" /> تأكيد التسوية</Button>
   892→          </DialogFooter>
   893→        </DialogContent>
   894→      </Dialog>
   895→
   896→      {/* Waybill print */}
   897→      <WaybillPrint shipment={shipment} open={waybillOpen} onOpenChange={setWaybillOpen} />
   898→    </div>
   899→  )
   900→}

   902→function Row({ label, value, ltr, money, bold }: { label: string; value: string; ltr?: boolean; money?: boolean; bold?: boolean }) {
   903→  return (
   904→    <div className="flex items-center justify-between gap-3 py-1 border-b last:border-0">
   905→      <span className="text-muted-foreground text-xs">{label}</span>
   906→      <span className={cn('text-right', ltr && 'dir-ltr font-mono', money && 'tnum', bold && 'font-bold')} dir={ltr ? 'ltr' : undefined}>{value}</span>
   907→    </div>
   908→  )
   909→}
