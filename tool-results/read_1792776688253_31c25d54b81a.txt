   396→    </Dialog>

   400→function ShipmentDetail({ id, onBack }: { id: string; onBack: () => void }) {
   401→  const { toast } = useToast()
   402→  const { data: shipment, mutate } = useSWR<any>(`/api/shipments/${id}`, fetcher)
   403→  const { items: statusItems } = useLookup('shipment_status')
   404→  const { items: collectionItems } = useLookup('collection_status')
   405→  const [statusOpen, setStatusOpen] = React.useState(false)
   406→  const [newStatus, setNewStatus] = React.useState('')
   407→  const [statusNotes, setStatusNotes] = React.useState('')
   408→
   409→  if (!shipment) return <Skeleton className="h-96 rounded-xl" />
   410→
   411→  const STATUS_LOOKUP: Record<string, { label: string; tone: string }> = Object.fromEntries(
   412→    statusItems.map((i) => [i.key, { label: i.label, tone: i.tone || '' }]),
   413→  )
   414→  const COLLECTION_LOOKUP: Record<string, { label: string; tone: string }> = Object.fromEntries(
   415→    collectionItems.map((i) => [i.key, { label: i.label, tone: i.tone || '' }]),
   416→  )
   417→  const sMeta = STATUS_LOOKUP[shipment.status] || { label: shipment.status, tone: '' }
   418→  const cMeta = COLLECTION_LOOKUP[shipment.collectionStatus] || { label: shipment.collectionStatus, tone: '' }
   419→
   420→  const changeStatus = async () => {
   421→    if (!newStatus) return
   422→    try {
   423→      await apiPost(`/api/shipments/${id}/status`, { status: newStatus, notes: statusNotes, userName: 'admin' })
   424→      toast({ title: 'تم تحديث الحالة' })
   425→      setStatusOpen(false); setNewStatus(''); setStatusNotes('')
   426→      mutate()
   427→    } catch (e: any) { toast({ title: 'خطأ', description: e.message, variant: 'destructive' }) }
   428→  }
   429→
   430→  const remove = async () => {
   431→    if (!confirm('هل تريد حذف هذه الشحنة؟ لا يمكن التراجع.')) return
   432→    try {
   433→      await apiDelete(`/api/shipments/${id}`)
   434→      toast({ title: 'تم الحذف' })
   435→      onBack()
   436→    } catch (e: any) { toast({ title: 'خطأ', description: e.message, variant: 'destructive' }) }
   437→  }
   438→
   439→  return (
   440→    <div className="space-y-4">
   441→      <div className="flex items-center justify-between">
   442→        <Button variant="ghost" size="sm" onClick={onBack}><ArrowLeft className="size-4" /> رجوع للقائمة</Button>
   443→        <div className="flex gap-2">
   444→          <Button variant="outline" size="sm" onClick={() => window.print()}><Printer className="size-4" /> طباعة بوليصة</Button>
   445→          <Button variant="outline" size="sm" onClick={() => setStatusOpen(true)}>تغيير الحالة</Button>
   446→          <Button variant="outline" size="sm" className="text-destructive" onClick={remove}><Trash2 className="size-4" /> حذف</Button>
   447→        </div>
   448→      </div>
   449→
   450→      <Card>
   451→        <CardContent className="p-5">
   452→          <div className="flex flex-col md:flex-row md:items-start md:justify-between gap-3">
   453→            <div>
   454→              <div className="text-xs text-muted-foreground">رقم الشحنة</div>
   455→              <div className="text-2xl font-bold font-mono">{shipment.shipmentNo}</div>
   456→              {shipment.billNo && <div className="text-sm text-muted-foreground tnum mt-1">بوليصة: {shipment.billNo}</div>}
   457→            </div>
   458→            <div className="flex gap-2">
   459→              <Badge variant="outline" className={cn('font-bold', sMeta.tone)}>{sMeta.label}</Badge>
   460→              <Badge variant="outline" className={cn('font-bold', cMeta.tone)}>{cMeta.label}</Badge>
   461→            </div>
   462→          </div>
   463→        </CardContent>
   464→      </Card>
   465→
   466→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   467→        <Card>
   468→          <CardContent className="p-4 space-y-2 text-sm">
   469→            <h3 className="font-semibold mb-2">بيانات العميل والتسليم</h3>
   470→            <Row label="العميل" value={shipment.customer?.name || shipment.recipientName || '-'} />
   471→            <Row label="جهة الشحن" value={shipment.shipToParty || '-'} />
   472→            <Row label="هاتف التوصيل" value={shipment.deliveryPhones || '-'} ltr />
   473→            <Row label="الوجهة" value={shipment.destination || '-'} />
   474→            <Row label="الوجهة (EN)" value={shipment.destinationEn || '-'} ltr />
   475→            <Row label="خط السير/التكويد" value={shipment.routeCode || '-'} />
   476→          </CardContent>
   477→        </Card>
   478→        <Card>
   479→          <CardContent className="p-4 space-y-2 text-sm">
   480→            <h3 className="font-semibold mb-2">بيانات التشغيل</h3>
   481→            <Row label="الفرع" value={shipment.branch?.name || '-'} />
   482→            <Row label="المخزن" value={shipment.warehouse?.name || '-'} />
   483→            <Row label="المقاول" value={shipment.contractor?.name || '-'} />
   484→            <Row label="نوع الحمولة" value={shipment.loadType || '-'} />

   486→            <Row label="الكمية (سياسات)" value={fmtNum(shipment.policyQty)} />
   487→          </CardContent>

   489→      </div>
   490→
   491→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">

   493→          <CardContent className="p-4 space-y-2 text-sm">

   496→            <Row label="إجمالي النولون" value={`${fmtNum(shipment.freightTotal)} ج.م`} money bold />

   505→          </CardContent>

   516→        </Card>

   518→
   519→      {/* Status history */}

   557→          </DialogFooter>

   560→    </div>
