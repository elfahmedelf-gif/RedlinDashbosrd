    29→// === Chart of Accounts ===
    30→export function AccountsModule() {
    31→  const { toast } = useToast()
    32→  const [open, setOpen] = React.useState(false)
    33→  const { data, isLoading, mutate } = useSWR<{ items: any[] }>('/api/accounts', fetcher, { refreshInterval: 30000 })
    34→
    35→  // build tree
    36→  const tree = React.useMemo(() => {
    37→    if (!data) return []
    38→    const byParent = new Map<string, any[]>()
    39→    for (const a of data.items) {
    40→      const pid = a.parentId || 'root'
    41→      if (!byParent.has(pid)) byParent.set(pid, [])
    42→      byParent.get(pid)!.push(a)
    43→    }
    44→    const build = (pid: string, depth: number): any[] => {
    45→      const children = byParent.get(pid) || []
    46→      return children.flatMap((c) => [{ ...c, depth }, ...build(c.id, depth + 1)])
    47→    }
    48→    return build('root', 0)
    49→  }, [data])
    50→
    51→  const fields: Field[] = [
    52→    { name: 'code', label: 'كود الحساب *', type: 'text' },
    53→    { name: 'name', label: 'اسم الحساب *', type: 'text' },
    54→    { name: 'nameEn', label: 'الاسم (EN)', type: 'text' },
    55→    { name: 'type', label: 'النوع *', type: 'select', options: Object.entries(ACCOUNT_TYPE).map(([v, m]) => ({ value: v, label: m.label })) },
    56→    { name: 'parentId', label: 'الحساب الأب', type: 'select', options: (data?.items ?? []).map((a) => ({ value: a.id, label: `${a.code} - ${a.name}` })) },
    57→    { name: 'isDetail', label: 'حساب تفصيلي (يقبل قيودًا)', type: 'switch' },
    58→  ]
    59→  const submit = async (vals: any) => {
    60→    await apiPost('/api/accounts', { ...vals, isDetail: vals.isDetail !== false, userName: 'admin' })
    61→    toast({ title: 'تمت الإضافة' }); mutate()
    62→  }
    63→
    64→  const columns = [
    65→    { key: 'code', header: 'الكود', align: 'center' as const, width: 'w-24', hideInCard: true, render: (a: any) => <span className="font-mono text-xs">{a.code}</span> },
    66→    { key: 'name', header: 'اسم الحساب', primary: true, render: (a: any) => (
    67→      <span style={{ paddingRight: `${a.depth * 16}px` }} className={cn('text-sm block truncate-1', a.isDetail ? 'font-medium' : 'font-bold text-primary')}>
    68→        {a.name}
    69→      </span>
    70→    ) },
    71→    { key: 'type', header: 'النوع', align: 'center' as const, width: 'w-32', render: (a: any) => {
    72→      const t = ACCOUNT_TYPE[a.type] || { label: a.type, tone: '' }
    73→      return <Badge variant="outline" className={t.tone}>{t.label}</Badge>
    74→    } },
    75→    { key: 'debit', header: 'مدين', align: 'left' as const, width: 'w-28', render: (a: any) => <span className="tnum">{fmtNum(a.debit)}</span> },
    76→    { key: 'credit', header: 'دائن', align: 'left' as const, width: 'w-28', render: (a: any) => <span className="tnum">{fmtNum(a.credit)}</span> },
    77→    { key: 'balance', header: 'الرصيد', align: 'left' as const, width: 'w-28', render: (a: any) => <span className="tnum font-bold">{fmtNum(a.balance)}</span> },
    78→  ]
    79→
    80→  return (
    81→    <div className="space-y-4">
    82→      <div className="flex items-center justify-between gap-2">
    83→        <div className="flex items-center gap-2 min-w-0">
    84→          <BookOpen className="size-5 text-muted-foreground shrink-0" />
    85→          <span className="text-sm text-muted-foreground tnum">{data?.items.length ?? 0} حساب</span>
    86→        </div>
    87→        <Button onClick={() => setOpen(true)} size="sm" className="shrink-0"><Plus className="size-4" /> <span className="hidden sm:inline">حساب جديد</span><span className="sm:hidden">جديد</span></Button>
    88→      </div>
    89→      <Card>
    90→        <CardHeader className="pb-2"><CardTitle className="text-base">شجرة الحسابات</CardTitle><CardDescription>مع أرصدة من القيود المرحّلة</CardDescription></CardHeader>
    91→        <CardContent className="p-0">
    92→          <DataTable rows={tree} columns={columns} loading={isLoading} getRowKey={(a: any) => a.id} minW={800} searchFn={(a, q) => (a.name || '').toLowerCase().includes(q) || (a.code || '').toLowerCase().includes(q)} searchPlaceholder="بحث بالاسم أو الكود..." />
    93→        </CardContent>
    94→      </Card>
    95→      <EntityFormDialog open={open} onOpenChange={setOpen} title="إضافة حساب" fields={fields} initial={{}} onSubmit={submit} />
    96→    </div>
    97→  )
    98→}
